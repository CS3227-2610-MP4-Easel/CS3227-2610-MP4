export const PlanTaskInputSchema = z.object({ title: z.string().min(1).max(200), description: z.string().max(4000), acceptanceCriteria: z.array(z.string().min(1).max(300)).max(12), storyPoints: z.union([z.literal(1), z.literal(2), z.literal(3), z.literal(5), z.literal(8), z.literal(13)]) }).strict();
export const PlanEpicEditBodySchema = z.object({ title: z.string().min(1).max(200).optional(), description: z.string().max(4000).optional(), checkpointName: z.string().max(200).nullable().optional(), tasks: z.array(PlanTaskInputSchema).min(1).max(8).optional(), version: <int 1..2147483647> }).strict()
  .refine((b) => b.title !== undefined || b.description !== undefined || b.checkpointName !== undefined || b.tasks !== undefined, { message: "Send at least one field." });
export const PlanCheckpointsBodySchema = z.object({ checkpoints: z.array(z.object({ name: z.string().min(1).max(200), targetDate: date, previousName: z.string().min(1).max(200).optional() }).strict()).max(8), version: <int> }).strict();
export const PlanEpicAddBodySchema = z.object({ title: z.string().min(1).max(200), description: z.string().max(4000), checkpointName: z.string().max(200).nullable(), version: <int> }).strict();
export const ProposalVersionQuerySchema = z.object({ version: z.coerce.number().int().min(1).max(2147483647) }).strict();
export const PlanListQuerySchema = PageQuerySchema.extend({ status: ProposalStatusSchema.optional(), size: z.coerce.number().int().min(1).max(20).default(20) });
export const PlanProposalPageSchema = pageSchema(PlanProposalSchema);
