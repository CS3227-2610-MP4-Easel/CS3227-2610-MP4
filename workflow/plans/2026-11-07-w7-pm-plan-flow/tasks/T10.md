```ts
export class PlanFlowError extends Error { constructor(readonly code: string, readonly status: number, message: string, readonly retryAfterSeconds?: number, readonly details?: { reason: string }) }
/** The proposals functions already bound to a database client, core and sessions (built in T11's `runtime.ts` and in test helpers), so this file never sees `FilteredDbClient`. */
export type ProposalsApi = {
  createProposal(i: CreateProposalInput): Promise<string>;
  getProposalView(proposalId: string, viewerId: string): Promise<PlanProposal | null>;
  claimEpic(i: { proposalId: string; index: number; actorId: string; now?: Date }): Promise<ClaimResult>;
  storeEpic(i: { proposalId: string; index: number; claimedVersion: number; tasks: EpicTask[]; flagged: boolean }): Promise<PlanEpic>;
  failEpic(i: { proposalId: string; index: number; claimedVersion: number; error: string }): Promise<void>;
  approveProposal(i: { proposalId: string; version: number; actorId: string }): Promise<PlanApproval>;
  rejectProposal(i: { proposalId: string; version: number; actorId: string }): Promise<PlanProposal>;
};
export type PlanFlowDeps = { proposals: ProposalsApi; gateway: LlmGateway; sessions: GatewaySessions; aiContext: AiContextWith; clock?: { now(): Date } };
export type PlanFlow = ReturnType<typeof createPlanFlow>;
export type Caller = { userId: string; role: "PM" };
export function createPlanFlow(deps: PlanFlowDeps): {
  createPlan(i: { caller: Caller; projectId: string; idea: string }): Promise<PlanProposal>;
  getProposal(i: { caller: Caller; proposalId: string }): Promise<PlanProposal>;
  expandEpic(i: { caller: Caller; proposalId: string; index: number }): Promise<PlanEpic>;
  approve(i: { caller: Caller; proposalId: string; version: number }): Promise<PlanApproval>;
  reject(i: { caller: Caller; proposalId: string; version: number }): Promise<PlanProposal>;
};
```
Gateway failure mapping (used by createPlan and expandEpic): `BLOCKED_INPUT` -> 422 `VALIDATION_FAILED`; `REQUEST_REFUSED` -> 422 `REQUEST_REFUSED`; `AI_BUSY` -> 429 `AI_BUSY`, `retryAfterSeconds` = result value or 60, `details.reason` `LOCAL_LIMIT`; `AI_UNAVAILABLE` -> 503 `AI_UNAVAILABLE`; `AI_DISABLED` -> 503 `AI_DISABLED`; `ROLE_DENIED`, `INVALID_LLM_CALL` and any unexpected error -> 500 `INTERNAL` (rethrow the original for T11 to map); `NotMember` -> 404 `NOT_FOUND`; thrown `NOT_FOUND` from the session check -> 404; thrown `CONFLICT_STATE` from the session check -> 409 `CONFLICT_STATE`, message "The AI session expired; start a new plan."; `SessionCapReached` (createPlan only) -> 429 `RATE_LIMITED` with its `retryAfterSeconds`.
`createPlan` order: (1) `sanitiseInput(idea).blocked` -> 422 `VALIDATION_FAILED` before anything else; (2) `sessions.open({ feature: "PM_PLAN_OUTLINE", userId, projectId })`; (3) in a try block (any failure from here closes the session `ABANDONED` with `sessions.close(sessionId, "ABANDONED", userId)` before the error propagates): `aiContext.assembleWith(caller, "PM_PLAN_OUTLINE", projectId, { query: idea })`; (4) `buildOutlineSections` with `today`; (5) `gateway.run({ feature: "PM_PLAN_OUTLINE", caller, sessionId, projectId, sections, output: { kind: "json", schema: outlineOutputSchema(today) }, inputHash })`; failure -> mapping; (6) `proposals.createProposal` with `retrieved` = provenance entries of section `similar-items` as `{ source: type, projectId, id }`, `flagged` = `result.flagged`; `ProposalError("CONFLICT_STATE")` -> 409 `CONFLICT_STATE` (project archived during the call); (7) return `getProposalView`.
`expandEpic` order: `claimEpic` (creator only; its `NOT_FOUND`/`CONFLICT_STATE` map to 404/409); `DONE` -> return the stored epic (no call, no session touch); then, in try: `assembleWith(caller, "PM_PLAN_EXPAND", projectId, { query: title + " " + description })`, `buildExpandSections`, `gateway.run({ feature: "PM_PLAN_EXPAND", sessionId: claim.sessionId, ... })`; success -> `storeEpic` (its `CONFLICT_STATE` -> 409 and the result is discarded); any failure after the claim -> `failEpic` with `error` = the gateway code (`AI_BUSY`, `AI_UNAVAILABLE`, `AI_DISABLED`, `REQUEST_REFUSED`, `BLOCKED_INPUT`), `SESSION_EXPIRED` for the session `CONFLICT_STATE`, `INTERNAL` for everything else including `NotMember`, then throw the mapped error. The session stays OPEN.
