## Contract
`[contract]` `schemas.ts` (use `import { z } from "zod/v3"` as contracts does; the gateway only needs an object with `safeParse` (`output-check.ts` `SafeParser`)):
```ts
export const POINTS = [1, 2, 3, 5, 8, 13] as const;
export function outlineOutputSchema(today: string): ZodTypeAny;  // today = YYYY-MM-DD in Asia/Singapore
// { approach: string 1-1500, checkpoints: {name 1-200, targetDate YYYY-MM-DD}[] 0-8, epics: {title 1-200, description 1-1000, checkpointName: string|null}[] 1-10 }
// refine: targetDate is a real calendar date, today <= date <= today + 730 days; checkpoint names unique ignoring case; each checkpointName is null or equals a checkpoint name exactly.
export const expandOutputSchema: ZodTypeAny;   // { tasks: {title 1-200, description 1-1000, acceptanceCriteria: string[] 1-6 each 1-300, storyPoints: one of POINTS}[] 1-8 }
```
`[contract]` `sections.ts`:
```ts
export const PM_PROMPT = { outline: { rules: 600, outputFormat: 500, today: 30, idea: 1750 }, expand: { rules: 500, outputFormat: 400, approach: 300, epic: 600 } } as const;  // budgets in tokens
export function outlineStaticSections(): LlmCall["sections"];             // rules + output-format, kind STATIC, priority 0, trusted true
export function buildOutlineSections(input: { today: string; idea: string; assembled: ContextSections }): ContextSections;
export function buildExpandSections(input: { approach: string; epic: { title: string; description: string }; assembled: ContextSections }): ContextSections;
export function inputHash(feature: Feature, sections: ContextSections): string;  // sha256 hex of feature + "\n" + sections.map(s => s.text).join("\u0000")
```
Outline order: `rules`, `output-format`, `today` (DYNAMIC, trusted true, priority 0, budget 30, text = the date), the assembled sections, `idea` (DYNAMIC, trusted false, priority 0, budget 1750). Expand order: `rules`, `output-format`, assembled sections, `approach` (DYNAMIC, trusted false, priority 0, budget 300), `epic` (DYNAMIC, trusted false, priority 0, budget 600, text `title: <title>\ndescription: <description>`).
